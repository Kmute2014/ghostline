/**
 * Next.js instrumentation hook â€” runs once per server boot.
 *
 * Boots the Ghostline realtime service (socket.io :3003 + internal notify
 * :3004) in-process so it shares the Next.js process lifetime. If the
 * standalone mini-service (mini-services/realtime) already owns the port,
 * this becomes a no-op â€” see the port guard in service.ts.
 *
 * Set DISABLE_INPROCESS_REALTIME=1 on hosts where realtime runs as its own
 * service (e.g. the Cloud Run production topology) so the web tier never boots a
 * second, unreachable instance.
 */
export async function register() {
  if (process.env.NEXT_RUNTIME !== 'nodejs') return
  if (process.env.DISABLE_INPROCESS_REALTIME === '1') return
  try {
    const { startRealtimeService } = await import('../mini-services/realtime/service')
    await startRealtimeService()
  } catch (err) {
    console.error('[rt] in-process realtime boot failed:', err)
  }
}

/**
 * Fire-and-forget notifier: pushes realtime events from Next.js API routes to
 * Firebase Firestore. The Socket.IO service remains a compatibility transport
 * for presence, typing, and socket-originated messages while the migration is
 * rolled out.
 */
import {
  publishFirebaseBroadcastEvent,
  publishFirebaseUserEvent,
} from '@/lib/firebase-admin'

const INTERNAL_URL = process.env.INTERNAL_NOTIFY_URL ?? 'http://127.0.0.1:3004/notify'
const INTERNAL_TOKEN = process.env.INTERNAL_TOKEN ?? 'ghostline-internal-dev'

async function postNotify(body: Record<string, unknown>, event: string): Promise<void> {
  try {
    const res = await fetch(INTERNAL_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-internal-token': INTERNAL_TOKEN },
      body: JSON.stringify(body),
      signal: AbortSignal.timeout(2000),
    })
    if (!res.ok) {
      console.warn(`[notify] service responded ${res.status} for event=${event}`)
    }
  } catch {
    console.warn(`[notify] service unreachable for event=${event} (clients will poll instead)`)
  }
}

export async function notifyUsers(userIds: string[], event: string, payload: unknown): Promise<void> {
  if (userIds.length === 0) return
  const [firebaseResult] = await Promise.allSettled([
    publishFirebaseUserEvent(userIds, event, payload),
    postNotify({ userIds, event, payload }, event),
  ])
  if (firebaseResult.status === 'rejected') {
    console.error(`[notify] Firebase publish failed for event=${event}:`, firebaseResult.reason)
  }
}

/**
 * Broadcast to a named socket.io room (e.g. the global "feed" room every
 * connected client joins). Used for Buzz feed events â€” everyone sees them.
 */
export async function notifyRoom(room: string, event: string, payload: unknown): Promise<void> {
  const [firebaseResult] = await Promise.allSettled([
    publishFirebaseBroadcastEvent(room, event, payload),
    postNotify({ rooms: [room], event, payload }, event),
  ])
  if (firebaseResult.status === 'rejected') {
    console.error(`[notify] Firebase publish failed for event=${event}:`, firebaseResult.reason)
  }
}

"use client";

import { io, type Socket } from 'socket.io-client'

let socket: Socket | null = null
let currentToken: string | null = null

/**
 * Singleton socket connection to the realtime service.
 *
 * - Production: set NEXT_PUBLIC_REALTIME_URL to the realtime service origin at
 *   BUILD time
 *   â€” NEXT_PUBLIC_ vars are inlined into the client bundle.
 * - Sandbox: unset â†’ connects same-origin through the Caddy gateway
 *   (XTransformPort=3003), which forwards the full path + query.
 *
 * The server serves engine.io under /rt/socket.io (see mini-services/realtime
 * service.ts) so one public port can also expose /healthz and /notify.
 */
export function getSocket(token: string): Socket {
  if (socket && currentToken === token) return socket
  if (socket) {
    socket.disconnect()
    socket = null
  }
  currentToken = token
  const realtimeUrl = process.env.NEXT_PUBLIC_REALTIME_URL
  socket = io(realtimeUrl || '/?XTransformPort=3003', {
    path: '/rt/socket.io',
    auth: { token },
    transports: ['websocket', 'polling'],
    forceNew: true,
    reconnection: true,
    reconnectionAttempts: Infinity,
    reconnectionDelay: 1000,
    reconnectionDelayMax: 5000,
    timeout: 10000,
  })
  return socket
}

export function disconnectSocket() {
  socket?.disconnect()
  socket = null
  currentToken = null
}

/** Current handshake token (or empty string when not initialized). */
export function currentSocketToken(): string {
  return currentToken ?? ''
}
