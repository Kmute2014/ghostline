import { NextRequest } from 'next/server'
import { FieldValue, Timestamp } from 'firebase-admin/firestore'
import { handle, ok, badRequest, forbidden, unauthorized } from '@/lib/api-helpers'
import { getAuth } from '@/lib/session'
import { getFirebaseAdminFirestore } from '@/lib/firebase-admin'
import { rateLimit } from '@/lib/rate-limit'
import { MAX_RADIUS_M } from '@/lib/geo'

export const dynamic = 'force-dynamic'

export const POST = handle(async (req: NextRequest) => {
  const auth = await getAuth()
  if (!auth) throw unauthorized()
  if (!rateLimit(`loc:${auth.profile.id}`, 30, 60_000)) throw badRequest('Location updates are too frequent.', 'RATE_LIMITED')
  const firestore = getFirebaseAdminFirestore()
  const profileRef = firestore.collection('profiles').doc(auth.profile.id)
  const profile = await profileRef.get()
  if (!profile.exists || !profile.data()?.nearbyVisible) throw forbidden('Enable nearby visibility before sharing location.', 'DISCOVERY_DISABLED')
  const body = await req.json().catch(() => ({}))
  const lat = Number(body?.lat)
  const lng = Number(body?.lng)
  const accuracy = body?.accuracy !== undefined && body?.accuracy !== null ? Number(body.accuracy) : null
  if (!Number.isFinite(lat) || !Number.isFinite(lng) || Math.abs(lat) > 90 || Math.abs(lng) > 180) throw badRequest('Invalid coordinates.', 'INVALID_COORDS')
  if (accuracy !== null && (!Number.isFinite(accuracy) || accuracy < 0 || accuracy > MAX_RADIUS_M * 4)) throw badRequest('Invalid accuracy value.', 'INVALID_COORDS')
  const now = Timestamp.now()
  await firestore.collection('locations').doc(auth.profile.id).set({ userId: auth.profile.id, lat, lng, accuracy, updatedAt: now }, { merge: true })
  return ok({ locationUpdatedAt: now.toDate().toISOString(), shared: true })
})

export const DELETE = handle(async () => {
  const auth = await getAuth()
  if (!auth) throw unauthorized()
  const firestore = getFirebaseAdminFirestore()
  const batch = firestore.batch()
  batch.delete(firestore.collection('locations').doc(auth.profile.id))
  batch.set(firestore.collection('profiles').doc(auth.profile.id), { nearbyVisible: false, updatedAt: FieldValue.serverTimestamp() }, { merge: true })
  await batch.commit()
  return ok({ shared: false, nearbyVisible: false })
})

export const PATCH = handle(async (req: NextRequest) => {
  const auth = await getAuth()
  if (!auth) throw unauthorized()
  const body = await req.json().catch(() => ({}))
  const data: Record<string, unknown> = { updatedAt: FieldValue.serverTimestamp() }
  const firestore = getFirebaseAdminFirestore()
  const profileRef = firestore.collection('profiles').doc(auth.profile.id)

  if (body.displayName !== undefined) {
    const trimmed = String(body.displayName).trim().replace(/\s+/g, ' ')
    if (!NAME_RE.test(trimmed)) throw badRequest('Display name must be 2â€“24 letters, numbers or spaces.', 'INVALID_NAME')
    data.displayName = trimmed
  }

  if (body.avatarFileId !== undefined) {
    const fileId = body.avatarFileId === null ? null : String(body.avatarFileId)
    if (fileId) {
      const fileSnap = await firestore.collection('files').doc(fileId).get()
      const file = fileSnap.data()
      if (!fileSnap.exists || file?.ownerId !== auth.profile.id || file?.kind !== 'avatar') throw badRequest('Invalid avatar upload.', 'INVALID_AVATAR')
    }
    const old = auth.profile.avatarFileId
    data.avatarFileId = fileId
    // The current project has media uploads disabled; clean only stale metadata.
    if (old && old !== fileId) {
      const oldRef = firestore.collection('files').doc(String(old))
      const oldSnap = await oldRef.get()
      if (oldSnap.exists && oldSnap.data()?.ownerId === auth.profile.id) await oldRef.delete()
    }
  }

  if (body.nearbyVisible !== undefined) {
    const visible = Boolean(body.nearbyVisible)
    data.nearbyVisible = visible
    if (!visible) await firestore.collection('locations').doc(auth.profile.id).delete()
  }
  if (body.showInPeople !== undefined) data.showInPeople = Boolean(body.showInPeople)
  if (body.notifyWhenPeopleOnline !== undefined) data.notifyWhenPeopleOnline = Boolean(body.notifyWhenPeopleOnline)
  if (body.onlineNotificationAudience !== undefined) {
    if (body.onlineNotificationAudience !== 'anyone' && body.onlineNotificationAudience !== 'friends') {
      throw badRequest('Choose anyone or friends for online notifications.', 'INVALID_ONLINE_NOTIFICATION_AUDIENCE')
    }
    data.onlineNotificationAudience = body.onlineNotificationAudience
  }
  if (body.defaultStoryAudience !== undefined) data.defaultStoryAudience = body.defaultStoryAudience === 'friends' ? 'friends' : 'public'

  await profileRef.update(data)
  const profileSnap = await profileRef.get()
  if (!profileSnap.exists) throw unauthorized()
  const [presenceSnap, locationSnap] = await Promise.all([
    firestore.collection('presences').doc(auth.profile.id).get(),
    firestore.collection('locations').doc(auth.profile.id).get(),
  ])
  const profile = { id: profileSnap.id, ...normalizeFirestoreValue(profileSnap.data()) }
  const presence = presenceSnap.exists ? normalizeFirestoreValue(presenceSnap.data()) : null
  const location = locationSnap.exists ? normalizeFirestoreValue(locationSnap.data()) : null
  return ok(toOwnProfile(profile as never, presence, location))
})

import { NextRequest } from 'next/server'
import { Timestamp } from 'firebase-admin/firestore'
import { handle, ok, badRequest, unauthorized } from '@/lib/api-helpers'
import { getAuth } from '@/lib/session'
import { getFirebaseAdminFirestore } from '@/lib/firebase-admin'
import { haversineMeters, boundingBox, approximateDistance, MAX_RADIUS_M } from '@/lib/geo'
import type { NearbyUser } from '@/lib/types'
import { avatarUrlFor } from '@/lib/dto'
import { blockedIdsFor, normalizeFirestoreValue } from '@/lib/firestore-route-helpers'

export const dynamic = 'force-dynamic'
export const LOCATION_FRESHNESS_MS = 15 * 60_000

export const GET = handle(async (req: NextRequest) => {
  const auth = await getAuth()
  if (!auth) throw unauthorized()
  const url = new URL(req.url)
  const radiusRaw = Number(url.searchParams.get('radius') ?? '1000')
  if (!Number.isFinite(radiusRaw) || radiusRaw < 5 || radiusRaw > MAX_RADIUS_M) throw badRequest(`Radius must be between 5 and ${MAX_RADIUS_M} metres.`, 'INVALID_RADIUS')
  const firestore = getFirebaseAdminFirestore()
  const [profileSnap, myLocSnap] = await Promise.all([
    firestore.collection('profiles').doc(auth.profile.id).get(),
    firestore.collection('locations').doc(auth.profile.id).get(),
  ])
  const me = profileSnap.data()
  const myLoc = myLocSnap.exists ? normalizeFirestoreValue(myLocSnap.data()) : null
  if (!me?.nearbyVisible || !myLoc) return ok({ users: [], requiresLocation: true })
  if (Date.now() - new Date(myLoc.updatedAt).getTime() > LOCATION_FRESHNESS_MS) return ok({ users: [], requiresLocation: true, stale: true })

  const box = boundingBox(Number(myLoc.lat), Number(myLoc.lng), radiusRaw)
  const cutoff = Timestamp.fromDate(new Date(Date.now() - LOCATION_FRESHNESS_MS))
  const [locSnap, blockedIds] = await Promise.all([
    firestore.collection('locations').where('updatedAt', '>', cutoff).limit(1000).get(),
    blockedIdsFor(firestore, auth.profile.id),
  ])
  const candidates = locSnap.docs.filter((doc) => {
    if (doc.id === auth.profile.id || blockedIds.has(doc.id)) return false
    const d = doc.data()
    return Number(d.lat) >= box.minLat && Number(d.lat) <= box.maxLat && Number(d.lng) >= box.minLng && Number(d.lng) <= box.maxLng
  })
  const users: NearbyUser[] = []
  for (const locDoc of candidates) {
    const [profileSnap2, presenceSnap] = await Promise.all([
      firestore.collection('profiles').doc(locDoc.id).get(),
      firestore.collection('presences').doc(locDoc.id).get(),
    ])
    if (!profileSnap2.exists || !presenceSnap.exists) continue
    const p = profileSnap2.data()!
    const presence = normalizeFirestoreValue(presenceSnap.data())
    if (!p.nearbyVisible || !presence.isOnline) continue
    const loc = locDoc.data()
    const distance = haversineMeters(Number(myLoc.lat), Number(myLoc.lng), Number(loc.lat), Number(loc.lng))
    if (distance > radiusRaw) continue
    const approx = approximateDistance(distance)
    users.push({
      publicId: String(p.publicId), displayName: String(p.displayName ?? 'Anonymous'),
      avatarUrl: avatarUrlFor(p.avatarFileId ?? null), online: true, lastSeenAt: null,
      distanceMeters: approx.meters, distanceLabel: approx.label,
    })
  }
  // Approximate distances may tie; coordinates never leave this handler.
  users.sort((a, b) => (a.distanceMeters ?? Infinity) - (b.distanceMeters ?? Infinity))
  return ok({ users: users.slice(0, 50), requiresLocation: false })
})

import { NextRequest } from 'next/server'
import { Timestamp } from 'firebase-admin/firestore'
import { handle, ok, unauthorized } from '@/lib/api-helpers'
import { getAuth } from '@/lib/session'
import { getFirebaseAdminFirestore } from '@/lib/firebase-admin'
import { isConversationExpired } from '@/lib/expire'
import { avatarUrlFor } from '@/lib/dto'
import type { OnlinePerson } from '@/lib/types'
import { blockedIdsFor, normalizeFirestoreValue } from '@/lib/firestore-route-helpers'

export const dynamic = 'force-dynamic'
const ONLINE_GRACE_MS = 75_000
const MAX_RESULTS = 200

export const GET = handle(async (_req: NextRequest) => {
  const auth = await getAuth()
  if (!auth) throw unauthorized()
  const firestore = getFirebaseAdminFirestore()
  const meId = auth.profile.id
  const cutoff = Timestamp.fromDate(new Date(Date.now() - ONLINE_GRACE_MS))
  const [presenceSnap, blockedIds, participationSnap, favoriteSnap] = await Promise.all([
    firestore.collection('presences').where('isOnline', '==', true).where('lastHeartbeatAt', '>', cutoff).orderBy('lastHeartbeatAt', 'desc').limit(MAX_RESULTS * 2).get(),
    blockedIdsFor(firestore, meId),
    firestore.collectionGroup('participants').where('userId', '==', meId).limit(500).get(),
    firestore.collection('contacts').where('ownerId', '==', meId).where('favorite', '==', true).get(),
  ])

  const convByOther = new Map<string, { id: string; updatedAt: Date }>()
  for (const part of participationSnap.docs) {
    const convRef = part.ref.parent.parent
    if (!convRef) continue
    const convSnap = await convRef.get()
    if (!convSnap.exists) continue
    const conv = normalizeFirestoreValue({ id: convSnap.id, ...convSnap.data() })
    if (conv.status !== 'active' || isConversationExpired(conv) || conv.type !== 'direct') continue
    const participants = await convRef.collection('participants').get()
    const other = participants.docs.find((d) => d.id !== meId)
    if (!other) continue
    const updatedAt = conv.updatedAt instanceof Date ? conv.updatedAt : new Date(conv.updatedAt ?? 0)
    const known = convByOther.get(other.id)
    if (!known || updatedAt > known.updatedAt) convByOther.set(other.id, { id: conv.id, updatedAt })
  }

  const favIds = new Set(favoriteSnap.docs.map((d) => String(d.data().contactId)))
  const users: OnlinePerson[] = []
  for (const presenceDoc of presenceSnap.docs) {
    const id = presenceDoc.id
    if (id === meId || blockedIds.has(id)) continue
    const profileSnap = await firestore.collection('profiles').doc(id).get()
    if (!profileSnap.exists) continue
    const profile = profileSnap.data()!
    if (profile.showInPeople !== true) continue
    users.push({
      publicId: String(profile.publicId), displayName: String(profile.displayName ?? 'Anonymous'),
      avatarUrl: avatarUrlFor(profile.avatarFileId ?? null), online: true, lastSeenAt: null,
      conversationId: convByOther.get(id)?.id ?? null, favorited: favIds.has(id),
    })
    if (users.length >= MAX_RESULTS) break
  }
  return ok({ users, count: users.length })
})

const NAME_RE = /^[\p{L}\p{N} _.'-]{2,24}$/u

function validateDisplayName(name: string | undefined): string {
  const trimmed = (name ?? '').trim().replace(/\s+/g, ' ')

  if (!trimmed) {
    return `Hush-${Math.floor(1000 + Math.random() * 9000)}`
  }

  if (!NAME_RE.test(trimmed)) {
    throw badRequest(
      'Display name must be 2â€“24 letters, numbers or spaces.',
      'INVALID_NAME',
    )
  }

  return trimmed
}

function ipKey(req: NextRequest): string {
  return (
    req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() ??
    req.headers.get('x-real-ip') ??
    'local'
  )
}

/**
 * Bootstrap check: does this browser already hold an anonymous identity?
 */
export const GET = handle(async () => {
  const auth = await getAuth()

  if (!auth) {
    return ok({
      authenticated: false,
      profile: null,
      socketToken: null,
      firebaseToken: null,
    })
  }

  const profile = auth.profile

  const [presence, location, session] = await Promise.all([
    getPresence(profile.id),
    getLocation(profile.id),
    getSessionByToken(auth.token),
  ])

  const firebaseToken = await createFirebaseCustomToken(profile.id)

  return ok({
    authenticated: true,
    profile: toOwnProfile(profile, presence, location),
    socketToken: session?.socketToken ?? null,
    firebaseToken,
  })
})

/**
 * Creates a new anonymous identity.
 * No email, phone number or password is required.
 */
export const POST = handle(async (req: NextRequest) => {
  if (!rateLimit(`signup:${ipKey(req)}`, 20, 60 * 60_000)) {
    throw tooMany(
      'Too many identities created from this network. Try later.',
      3600,
    )
  }

  const body = await req.json().catch(() => ({}))
  const displayName = validateDisplayName(body?.displayName)

  const profile = await createAnonymousIdentity(displayName)

  if (!profile) {
    throw new ApiError(500, 'INTERNAL', 'Could not create identity.')
  }

  const session = await createSessionForUser(profile.id)
  const firebaseToken = await createFirebaseCustomToken(profile.id)
  const presence = await getPresence(profile.id)

  const res = NextResponse.json({
    authenticated: true,
    isNewUser: true,
    profile: toOwnProfile(profile, presence, null),
    socketToken: session.socketToken,
    firebaseToken,
  })

  return attachSessionCookie(res, session.token)
})

/**
 * Optional identity recovery: binds this browser to an existing identity
 * using the recovery code shown in Profile â†’ Identity recovery.
 */
export const PUT = handle(async (req: NextRequest) => {
  if (!rateLimit(`recover:${ipKey(req)}`, 10, 60 * 60_000)) {
    throw tooMany(
      'Too many recovery attempts. Try again later.',
      3600,
    )
  }

  const body = await req.json().catch(() => ({}))
  const code = normalizeCode(String(body?.recoveryCode ?? ''))

  if (!isValidCodeShape(code, 12, 16)) {
    throw badRequest(
      'That recovery code does not look right.',
      'INVALID_RECOVERY_CODE',
    )
  }

  // Accept both the normalized code and its dashed stored format.
  const candidates = [code]

  if (code.length === 15) {
    candidates.unshift(
      `R${code.slice(1, 5)}-${code.slice(5, 10)}-${code.slice(10, 15)}`,
    )
  }

  const profile = await findProfileByRecoveryCode(candidates)

  if (!profile) {
    throw badRequest(
      'No identity matches this recovery code.',
      'RECOVERY_NOT_FOUND',
    )
  }

  const session = await createSessionForUser(profile.id)
  const firebaseToken = await createFirebaseCustomToken(profile.id)
  const presence = await getPresence(profile.id)

  const res = NextResponse.json({
    authenticated: true,
    recovered: true,
    profile: toOwnProfile(profile, presence, null),
    socketToken: session.socketToken,
    firebaseToken,
  })

  return attachSessionCookie(res, session.token)
})

// Isolated composer: owns its own text state so every keystroke re-renders only
// this component â€” not the whole message list, dialogs and sheets above it. It
// also owns the typing-emit throttle and the auto-grow height logic.
function ChatComposer({ conversationId, disabled = false, replyTo, onClearReply, onSendText, onSendMedia }: ChatComposerProps) {
  const reduced = useReducedMotion() ?? false
  const [text, setText] = useState('')
  const [sending, setSending] = useState(false)
  const textareaRef = useRef<HTMLTextAreaElement | null>(null)
  const imageInputRef = useRef<HTMLInputElement | null>(null)
  const videoInputRef = useRef<HTMLInputElement | null>(null)
  const lastTypingSent = useRef(0)
  const typingTimeout = useRef<ReturnType<typeof setTimeout> | null>(null)
  const sendingRef = useRef(false) // same-burst guard: rapid repeated Enter cannot double-send

  // Auto-grow like WhatsApp. useLayoutEffect runs before paint (no one-frame-late
  // jump), and field-sizing-fixed neutralizes the shadcn base field-sizing-content
  // so only the JS owns the height â€” the two sizing mechanisms used to fight.
  useLayoutEffect(() => {
    const el = textareaRef.current
    if (!el) return
    el.style.height = 'auto'
    el.style.height = `${Math.min(el.scrollHeight, 128)}px`
  }, [text])

  function emitTyping(typing: boolean) {
    try {
      getSocket(currentSocketToken()).emit('typing', {
        conversationId,
        typing,
      })
    } catch {
      /* socket not ready */
    }
  }

  function onInputChange(value: string) {
    setText(value)
    const now = Date.now()
    if (now - lastTypingSent.current > 1500) {
      lastTypingSent.current = now
      emitTyping(true)
    }
    if (typingTimeout.current) clearTimeout(typingTimeout.current)
    typingTimeout.current = setTimeout(() => emitTyping(false), 2000)
  }

  function sendText() {
    if (sendingRef.current) return
    const value = text.trim()
    if (!value) return
    sendingRef.current = true
    setSending(true)
    onSendText(value, replyTo?.id)
    setText('')
    if (replyTo) onClearReply()
    emitTyping(false)
    // Release the guard on the next macrotask so two Enters landing in the same
    // burst cannot send twice â€” the optimistic send itself stays synchronous.
    setTimeout(() => {
      sendingRef.current = false
      setSending(false)
    }, 0)
  }

  if (disabled) return null

  return (
    <div className="shrink-0 border-t border-border/70 bg-background/90 p-2.5 pb-[max(10px,env(safe-area-inset-bottom))] backdrop-blur-xl">

            <X className="h-4 w-4" aria-hidden />
          </button>
        </div>
      )}
      <div className="mx-auto flex max-w-4xl items-end gap-1.5 sm:gap-2">

          ref={imageInputRef}
          type="file"
          accept="image/jpeg,image/png,image/webp,image/gif"

          )}
          {m.replyTo && (
            <button
              type="button"
              className={cn('mb-1.5 block w-full rounded-md border-l-2 px-2 py-1 text-left', mine ? 'border-primary-foreground/50 bg-black/10' : 'border-primary bg-black/5')}
              onClick={() => document.getElementById(`message-${m.replyTo?.id}`)?.scrollIntoView({ behavior: 'smooth', block: 'center' })}
            >
              <span className="block text-[10px] font-semibold">
                {m.replyTo.senderPublicId === viewerPublicId ? 'You' : m.replyTo.senderName ?? m.replyTo.senderPublicId}
              </span>
              <span className="block truncate text-xs opacity-80">
                {m.replyTo.deletedForEveryone
                  ? 'This message was deleted'
                  : m.replyTo.text ?? (m.replyTo.type === 'image' ? 'Photo' : m.replyTo.type === 'video' ? 'Video' : 'Message')}
              </span>
            </button>
          )}
          {m.deletedForEveryone ? (
            <p className="italic opacity-75">This message was deleted</p>
          ) : m.type === 'text' && <p className="whitespace-pre-wrap break-words">{m.text}</p>}
          {!m.deletedForEveryone && m.type === 'image' && m.mediaUrl && (
            <button
              className="block -mx-1 -my-1"
              onClick={() => onLightbox(m)}
              aria-label="View image full size"
            >
              <img
                src={m.mediaUrl}
                alt="Shared photo"
                className="rounded-xl max-w-[220px] sm:max-w-[260px] max-h-[280px] object-cover"
              />
            </button>
          )}
          {!m.deletedForEveryone && m.type === 'video' && m.mediaUrl && (
            <video
              src={m.mediaUrl}
              controls
              playsInline
              preload="metadata"
              className="rounded-xl max-w-[220px] sm:max-w-[280px] max-h-[300px]"
            />
          )}
          <div className={cn('flex items-center gap-1 mt-0.5 justify-end', mine ? 'text-primary-foreground/70' : 'text-muted-foreground')}>
            <span className="text-[10px]">{shortTime(m.createdAt)}</span>
            {m.editedAt && !m.deletedForEveryone && <span className="text-[10px] italic">edited</span>}
            {mine &&
              (m.failed ? (
                <button onClick={() => onRetry(m)} aria-label="Retry sending" title="Failed â€” tap to retry">
                  <AlertCircle className="h-3 w-3 text-destructive" />
                </button>
              ) : m.pending ? (
                <span className="inline-flex items-center gap-1">
                  <Clock className="h-3 w-3 animate-pulse" aria-hidden />
                  <DeliveryLabel createdAt={m.createdAt} />
                </span>
              ) : readByOther ? (
                <CheckCheck className="h-3 w-3" aria-label="Read" />
              ) : (
                <Check className="h-3 w-3" aria-label="Sent" />
              ))}
          </div>
        </div>
      </div>
    </motion.div>
  )
})

export function ChatView({ conversationId }: { conversationId: string }) {
  const queryClient = useQueryClient()
  const setTab = useAppStore((s) => s.setTab)
  const typingState = useAppStore((s) => s.typing[conversationId])
  const reduced = useReducedMotion() ?? false

  const [lightbox, setLightbox] = useState<MessageDTO | null>(null)
  const [settingsOpen, setSettingsOpen] = useState(false)
  const [reportOpen, setReportOpen] = useState(false)
  const [pending, setPending] = useState<MessageDTO[]>([])
  const [renaming, setRenaming] = useState('')
  const [addMembersOpen, setAddMembersOpen] = useState(false)
  const [inviteInfo, setInviteInfo] = useState<{ joinCode: string; shareUrl: string } | null>(null)
  const [inviteLoading, setInviteLoading] = useState(false)
  const [rotating, setRotating] = useState(false)
  const [favoriteBusy, setFavoriteBusy] = useState(false)
  const [replyingTo, setReplyingTo] = useState<MessageDTO | null>(null)
  const [actionMessage, setActionMessage] = useState<MessageDTO | null>(null)
  const [editingMessage, setEditingMessage] = useState<MessageDTO | null>(null)
  const [editText, setEditText] = useState('')
  const [savingEdit, setSavingEdit] = useState(false)
  const [deletingMode, setDeletingMode] = useState<'me' | 'everyone' | null>(null)
  const [deletingMessage, setDeletingMessage] = useState<MessageDTO | null>(null)
  const [deletingBusy, setDeletingBusy] = useState(false)

  const bottomRef = useRef<HTMLDivElement | null>(null)
  const scrollRef = useRef<HTMLDivElement | null>(null)
  const pinnedRef = useRef(true) // list pinned to bottom until the user scrolls up
  const lastReadRef = useRef<number>(0)
  const lastViewportHeightRef = useRef(-1)
  const mediaFileIdsRef = useRef(new Map<string, string>())
  const mediaPreviewUrlsRef = useRef(new Map<string, string>())

  useEffect(
    () => () => {
      for (const url of mediaPreviewUrlsRef.current.values()) URL.revokeObjectURL(url)
    },
    []
  )

  const conversation = useQuery({
    queryKey: ['conversation', conversationId],
    queryFn: () => api.getConversation(conversationId),
    retry: false,
  })

  const messages = useInfiniteQuery({
    queryKey: ['messages', conversationId],
    queryFn: async ({ pageParam }: { pageParam: string | null }) =>
      api.getMessages(conversationId, pageParam ?? null),
    initialPageParam: null as string | null,
    getNextPageParam: () => undefined, // we page backwards only
    getPreviousPageParam: (firstPage) => (firstPage.hasMore ? firstPage.oldestCursor : undefined),
    maxPages: 20,
  })

  const conv = conversation.data?.conversation
  const blocked = conversation.data
    ? { me: conversation.data.iBlockedThem, them: conversation.data.theyBlockedMe }
    : { me: false, them: false }

  const gone =
    conversation.error instanceof ApiClientError &&
    (conversation.error.code === 'CONVERSATION_EXPIRED' || conversation.error.code === 'CONVERSATION_ENDED')

  const realMessages = useMemo(() => {
    const pages = messages.data?.pages ?? []
    return [...pages].reverse().flatMap((p) => p.messages)
  }, [messages.data])

  const otherLastReadAt = useMemo(() => {
    const pages = messages.data?.pages ?? []
    return pages[0]?.otherLastReadAt ?? null
  }, [messages.data])

  const myPublicId = useAppStore((s) => s.profile?.publicId)
  const allMessages = useMemo(() => {
    // Dedupe optimistic entries once the real message (same clientId) arrives
    const realClientIds = new Set(realMessages.map((m) => m.clientId).filter(Boolean))
    const pendingVisible = pending.filter((p) => !realClientIds.has(p.clientId))
    return [...realMessages, ...pendingVisible]
  }, [realMessages, pending])

  // Auto-scroll to bottom when messages change (if user is near the bottom)
  const scrollToBottom = useCallback((smooth = false) => {
    bottomRef.current?.scrollIntoView({ behavior: smooth ? 'smooth' : 'auto', block: 'end' })
  }, [])

  // Auto-scroll on new messages â€” but only while the user is still pinned to
  // the bottom: never yank the viewport while they are reading history, and
  // "Load older messages" prepending a page must not ruin the scroll position.
  useEffect(() => {
    if (pinnedRef.current) scrollToBottom()
  }, [allMessages.length, scrollToBottom])

  // Track whether the user is pinned to the bottom (WhatsApp behavior: opening
  // or closing the keyboard keeps you glued to the newest message).
  const markScrollPinned = useCallback(() => {
    const el = scrollRef.current
    if (!el) return
    pinnedRef.current = el.scrollHeight - el.scrollTop - el.clientHeight < 80
  }, [])

  // Use the visible viewport directly so browsers that already resize 100dvh
  // when the keyboard opens do not have the keyboard height subtracted twice.
  useEffect(() => {
    const vv = window.visualViewport
    if (!vv) return
    const root = document.documentElement
    let frame = 0
    const apply = () => {
      const overlap = Math.max(0, Math.round(window.innerHeight - vv.height - vv.offsetTop))
      const viewportHeight = Math.round(vv.height + vv.offsetTop)
      if (viewportHeight !== lastViewportHeightRef.current) {
        lastViewportHeightRef.current = viewportHeight
        root.style.setProperty('--gl-visual-viewport-height', `${viewportHeight}px`)
      }
      if (overlap > 0) window.scrollTo(0, 0) // keyboard opening â€” stop iOS window panning
      if (pinnedRef.current) scrollToBottom()
    }
    const schedule = () => {
      if (frame) return
      frame = requestAnimationFrame(() => {
        frame = 0
        apply()
      })
    }
    schedule()
    vv.addEventListener('resize', schedule)
    vv.addEventListener('scroll', schedule)
    return () => {
      vv.removeEventListener('resize', schedule)
      vv.removeEventListener('scroll', schedule)
      if (frame) cancelAnimationFrame(frame)
      root.style.removeProperty('--gl-visual-viewport-height')
      lastViewportHeightRef.current = -1
    }
  }, [scrollToBottom])

  // Mark read whenever the view is open, visible and messages exist
  useEffect(() => {
    if (!conv || allMessages.length === 0) return
    const now = Date.now()
    if (now - lastReadRef.current < 3000) return
    lastReadRef.current = now
    if (document.visibilityState === 'visible') {
      api.markRead(conversationId).catch(() => undefined)
    }
  }, [conv, allMessages.length, conversationId])

  // Ensure realtime subscription for this room (e.g. conversation created elsewhere)
  useEffect(() => {
    try {
      getSocket(currentSocketToken()).emit('conversation:subscribe', { conversationId })
    } catch {
      /* socket not initialized yet â€” rooms auto-join on connect */
    }
  }, [conversationId])

  const optimisticAppend = useCallback((m: MessageDTO) => {
    setPending((p) => [...p, m])
  }, [])

  const markOptimistic = useCallback((clientId: string, patch: Partial<MessageDTO>) => {
    setPending((p) => p.map((m) => (m.clientId === clientId ? { ...m, ...patch } : m)))
  }, [])

  const confirmOptimistic = useCallback(
    (clientId: string) => {
      markOptimistic(clientId, { pending: false, failed: false })
      mediaFileIdsRef.current.delete(clientId)
      const previewUrl = mediaPreviewUrlsRef.current.get(clientId)
      if (previewUrl) URL.revokeObjectURL(previewUrl)
      mediaPreviewUrlsRef.current.delete(clientId)
      void queryClient.invalidateQueries({ queryKey: ['messages', conversationId] }).then(() => {
        const latest = queryClient.getQueryData<typeof messages.data>(['messages', conversationId])
        if (latest?.pages.some((page) => page.messages.some((m) => m.clientId === clientId))) {
          setPending((p) => p.filter((m) => m.clientId !== clientId))
        }
      })
    },
    [conversationId, markOptimistic, messages.data, queryClient]
  )

  const sendViaHttp = useCallback(
    async (payload: { type: string; text?: string; fileId?: string | null; replyToId?: string }, clientId: string) => {
      try {
        await api.sendMessage(conversationId, { ...payload, clientId })
        // Keep the optimistic bubble visible until the refreshed history
        // contains the server copy; otherwise the message briefly disappears.
        confirmOptimistic(clientId)
        queryClient.invalidateQueries({ queryKey: ['conversations'] })
      } catch (err) {
        if (err instanceof ApiClientError && (err.code === 'CONVERSATION_EXPIRED' || err.code === 'CONVERSATION_ENDED')) {
          setPending((p) => p.filter((m) => m.clientId !== clientId))
          mediaFileIdsRef.current.delete(clientId)
          const previewUrl = mediaPreviewUrlsRef.current.get(clientId)
          if (previewUrl) URL.revokeObjectURL(previewUrl)
          mediaPreviewUrlsRef.current.delete(clientId)
          toast.error(err.message)
          queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
        } else if (err instanceof ApiClientError && err.code === 'INVALID_REPLY') {
          setPending((p) => p.filter((m) => m.clientId !== clientId))
          mediaFileIdsRef.current.delete(clientId)
          const previewUrl = mediaPreviewUrlsRef.current.get(clientId)
          if (previewUrl) URL.revokeObjectURL(previewUrl)
          mediaPreviewUrlsRef.current.delete(clientId)
          toast.error(err.message)
        } else {
          markOptimistic(clientId, { pending: false, failed: true })
        }
      }
    },
    [confirmOptimistic, conversationId, markOptimistic, queryClient]
  )

  const sendViaSocket = useCallback(
    (payload: { type: string; text?: string; fileId?: string | null; replyToId?: string }, clientId: string) => {
      const socket = getSocket(currentSocketToken())
      if (!socket.connected) {
        sendViaHttp(payload, clientId)
        return
      }
      socket.emit(
        'message:send',
        { conversationId, clientId, type: payload.type, text: payload.text, fileId: payload.fileId ?? undefined, replyToId: payload.replyToId },
        (ack: { ok?: boolean; message?: MessageDTO; error?: string }) => {
          if (ack?.ok && ack.message) {
            // The local copy bridges the ack-to-query-refetch gap and is
            // deduplicated as soon as history returns this clientId.
            confirmOptimistic(clientId)
            queryClient.invalidateQueries({ queryKey: ['conversations'] })
          } else if (ack?.error === 'conversation_expired' || ack?.error === 'not_participant') {
            setPending((p) => p.filter((m) => m.clientId !== clientId))
            mediaFileIdsRef.current.delete(clientId)
            const previewUrl = mediaPreviewUrlsRef.current.get(clientId)
            if (previewUrl) URL.revokeObjectURL(previewUrl)
            mediaPreviewUrlsRef.current.delete(clientId)
            toast.error('This conversation is closed and cannot accept messages.')
            queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
            queryClient.invalidateQueries({ queryKey: ['conversations'] })
          } else if (ack?.error === 'invalid_reply') {
            setPending((p) => p.filter((m) => m.clientId !== clientId))
            mediaFileIdsRef.current.delete(clientId)
            const previewUrl = mediaPreviewUrlsRef.current.get(clientId)
            if (previewUrl) URL.revokeObjectURL(previewUrl)
            mediaPreviewUrlsRef.current.delete(clientId)
            toast.error('That message is no longer available to reply to.')
          } else {
            markOptimistic(clientId, { pending: false, failed: true })
          }
        }
      )
    },
    [confirmOptimistic, conversationId, markOptimistic, queryClient, sendViaHttp]
  )

  // The text arrives fully trimmed from ChatComposer (which also owns the
  // `sending` double-send guard) â€” this stays a pure optimistic fire-and-forget.
  const sendText = useCallback(
    (value: string, replyToId?: string) => {
      const trimmed = value.trim()
      if (!trimmed) return
      const clientId = crypto.randomUUID()
      const quoted = replyToId ? allMessages.find((message) => message.id === replyToId) : null
      const replyTo = quoted
        ? {
            id: quoted.id,
            senderPublicId: quoted.senderPublicId,
            senderName: replyingTo?.id === quoted.id ? replyingTo.senderName ?? null : quoted.senderName ?? null,
            type: quoted.type,
            text: quoted.text,
            deletedForEveryone: Boolean(quoted.deletedForEveryone),
          }
        : null
      optimisticAppend({
        id: clientId,
        conversationId,
        senderPublicId: 'me',
        type: 'text',
        text: trimmed,
        mediaUrl: null,
        createdAt: new Date().toISOString(),
        clientId,
        pending: true,
        replyTo,
      })
      sendViaSocket({ type: 'text', text: trimmed, replyToId }, clientId)
    },
    [allMessages, conversationId, optimisticAppend, replyingTo, sendViaSocket]
  )

  const sendMedia = useCallback(
    async (file: File, kind: 'image' | 'video', replyToId?: string) => {
      const max = kind === 'image' ? 8 * 1024 * 1024 : 50 * 1024 * 1024
      if (file.size > max) {
        toast.error(`${kind === 'image' ? 'Images' : 'Videos'} must be under ${max / 1024 / 1024} MB.`)
        return
      }
      const clientId = crypto.randomUUID()
      const quoted = replyToId ? allMessages.find((message) => message.id === replyToId) : null
      const previewUrl = URL.createObjectURL(file)
      mediaPreviewUrlsRef.current.set(clientId, previewUrl)
      optimisticAppend({
        id: clientId,
        conversationId,
        senderPublicId: 'me',
        type: kind,
        text: null,
        mediaUrl: previewUrl,
        createdAt: new Date().toISOString(),
        clientId,
        pending: true,
        replyTo: quoted
          ? {
              id: quoted.id,
              senderPublicId: quoted.senderPublicId,
              senderName: replyingTo?.id === quoted.id ? replyingTo.senderName ?? null : quoted.senderName ?? null,
              type: quoted.type,
              text: quoted.text,
              deletedForEveryone: Boolean(quoted.deletedForEveryone),
            }
          : null,
      })
      try {
        const up = await api.upload(file, kind)
        mediaFileIdsRef.current.set(clientId, up.id)
        sendViaSocket({ type: kind, fileId: up.id, replyToId }, clientId)
        if (replyingTo?.id === replyToId) setReplyingTo(null)
      } catch (err) {
        setPending((p) => p.filter((m) => m.clientId !== clientId))
        mediaFileIdsRef.current.delete(clientId)
        const previewUrl = mediaPreviewUrlsRef.current.get(clientId)
        if (previewUrl) URL.revokeObjectURL(previewUrl)
        mediaPreviewUrlsRef.current.delete(clientId)
        toast.error(err instanceof ApiClientError ? err.message : 'Upload failed.')
      }
    },
    [allMessages, conversationId, optimisticAppend, replyingTo, sendViaSocket]
  )

  const retryMessage = useCallback(
    (m: MessageDTO) => {
      const clientId = m.clientId ?? crypto.randomUUID()
      const fileId = mediaFileIdsRef.current.get(clientId)
      if (m.type !== 'text' && !fileId) {
        toast.error('This attachment is no longer available to retry. Please select it again.')
        return
      }
      setPending((p) =>
        p.map((x) =>
          x.id === m.id ? { ...x, clientId, pending: true, failed: false } : x
        )
      )
      if (m.type === 'text') {
        sendViaSocket({ type: 'text', text: m.text ?? '' }, clientId)
      } else {
        sendViaSocket({ type: m.type, fileId }, clientId)
      }
    },
    [sendViaSocket]
  )

  // Stable callbacks handed down to every memoized MessageBubble.
  const openLightbox = useCallback((m: MessageDTO) => setLightbox(m), [])
  const openMessageActions = useCallback((m: MessageDTO) => setActionMessage(m), [])

  async function saveMessageEdit() {
    if (!editingMessage || !editText.trim()) return
    setSavingEdit(true)
    try {
      await api.editMessage(conversationId, editingMessage.id, editText)
      setEditingMessage(null)
      setEditText('')
      queryClient.invalidateQueries({ queryKey: ['messages', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      toast.success('Message updated.')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not edit the message.')
    } finally {
      setSavingEdit(false)
    }
  }

  async function confirmMessageDelete() {
    if (!deletingMessage || !deletingMode) return
    setDeletingBusy(true)
    try {
      await api.deleteMessage(conversationId, deletingMessage.id, deletingMode)
      setDeletingMode(null)
      setDeletingMessage(null)
      queryClient.invalidateQueries({ queryKey: ['messages', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      toast.success(deletingMode === 'everyone' ? 'Message deleted for everyone.' : 'Message deleted for you.')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not delete the message.')
    } finally {
      setDeletingBusy(false)
    }
  }

  async function unblock() {
    if (!conv?.other) return
    try {
      await api.unblockUser(conv.other.publicId)
      toast.success('Unblocked.')
      queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['blocks'] })
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not unblock.')
    }
  }

  async function changeExpiration(policy: ExpirationPolicyDTO) {
    try {
      await api.updateConversation(conversationId, { expirationPolicy: policy })
      queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      toast.success('Expiration updated for everyone in this chat.')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not update expiration.')
    }
  }

  async function saveNickname() {
    if (!conv?.other) return
    const value = renaming.trim()
    if (!value) return
    try {
      await api.renameContact(conv.other.publicId, value)
      toast.success(`Now showing them as "${value}" â€” only you see this.`)
      queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      queryClient.invalidateQueries({ queryKey: ['contacts'] })
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not rename.')
    }
  }

  async function saveAsContact() {
    if (!conv?.other) return
    try {
      await api.saveContact(conv.other.publicId)
      toast.success('Saved as a contact. They can now see your friends-only stories.')
      queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['contacts'] })
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not save contact.')
    }
  }

  async function toggleFavorite() {
    if (!conv?.other || favoriteBusy) return
    const next = !conv.other.favorited
    setFavoriteBusy(true)
    try {
      await api.setContactFavorite(conv.other.publicId, next)
      toast.success(
        next
          ? `${conv.other.displayName} added to favourites.`
          : `${conv.other.displayName} removed from favourites.`
      )
      queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      queryClient.invalidateQueries({ queryKey: ['contacts'] })
      queryClient.invalidateQueries({ queryKey: ['people'] })
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not update favourite.')
    } finally {
      setFavoriteBusy(false)
    }
  }

  async function block() {
    if (!conv?.other) return
    try {
      await api.blockUser(conv.other.publicId)
      toast.success('Blocked. They can no longer contact you.')
      queryClient.invalidateQueries({ queryKey: ['conversation', conversationId] })
      queryClient.invalidateQueries({ queryKey: ['blocks'] })
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not block.')
    }
  }

  async function endConversation() {
    try {
      await api.updateConversation(conversationId, { end: true })
      toast.success('Conversation ended.')
      setSettingsOpen(false)
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      setTab('chats')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not end conversation.')
    }
  }

  async function deleteConversation() {
    try {
      await api.deleteConversation(conversationId)
      toast.success('Conversation deleted for everyone.')
      setSettingsOpen(false)
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      setTab('chats')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not delete conversation.')
    }
  }

  async function leaveGroup() {
    try {
      await api.leaveGroup(conversationId)
      toast.success('You left the group.')
      setSettingsOpen(false)
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      setTab('chats')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not leave the group.')
    }
  }

  async function loadInvite(rotate: boolean) {
    setInviteLoading(true)
    setRotating(rotate)
    try {
      const res = await api.getGroupInvite(conversationId, rotate)
      setInviteInfo({ joinCode: res.joinCode, shareUrl: res.shareUrl })
      if (rotate) toast.success('New code generated â€” the old link no longer works.')
    } catch (err) {
      toast.error(err instanceof ApiClientError ? err.message : 'Could not load the invite code.')
    } finally {
      setInviteLoading(false)
      setRotating(false)
    }
  }

  // Join codes are per-conversation â€” drop the cached one when switching chats.
  useEffect(() => {
    setInviteInfo(null)
  }, [conversationId])

/** Requests browser geolocation (permission prompt) and shares it. */
export async function shareCurrentLocation(): Promise<{ ok: boolean; error?: string }> {
  if (!('geolocation' in navigator)) return { ok: false, error: 'This browser does not support location.' }
  return new Promise((resolve) => {
    navigator.geolocation.getCurrentPosition(
      async (pos) => {
        try {
          await api.updateLocation(pos.coords.latitude, pos.coords.longitude, pos.coords.accuracy)
          resolve({ ok: true })
        } catch (err) {
          resolve({ ok: false, error: err instanceof ApiClientError ? err.message : 'Could not share location.' })
        }
      },
      (err) => {
        const msg =
          err.code === err.PERMISSION_DENIED
            ? 'Location permission was denied. Enable it in your browser settings to use Nearby.'
            : 'Could not determine your location. Try again.'
        resolve({ ok: false, error: msg })
      },
      { enableHighAccuracy: true, timeout: 12_000, maximumAge: 30_000 }
    )
  })
}

/** Promisified geolocation with high accuracy. */
function getCurrentPosition(): Promise<GeolocationPosition> {
  return new Promise((resolve, reject) => {
    if (!('geolocation' in navigator)) {
      reject(new Error('GEO_UNSUPPORTED'))
      return
    }
    navigator.geolocation.getCurrentPosition(
      resolve,
      (err) => {
        if (err.code === err.PERMISSION_DENIED) reject(new Error('GEO_DENIED'))
        else reject(new Error('GEO_ERROR'))
      },
      { enableHighAccuracy: true, timeout: 12_000, maximumAge: 30_000 },
    )
  })
}

    const onChange = () => {
      if (!cancelled && status) setPermission(status.state as PermissionView)
    }
    async function check() {
      try {
        if (!navigator.permissions?.query) {
          setPermission('unsupported')
          return
        }
        status = await navigator.permissions.query({ name: 'geolocation' })
        if (cancelled) return
        onChange()
        status.addEventListener('change', onChange)
      } catch {
        if (!cancelled) setPermission('unsupported')
      }
    }
    void check()
    return () => {
      cancelled = true
      status?.removeEventListener('change', onChange)
    }
  }, [])

  async function handleNearbyChange(next: boolean) {
    if (locationBusy) return
    setLocationBusy(true)
    if (next) {
      setNearbyLocal(true) // optimistic; revert on failure
      try {
        const position = await getCurrentPosition()
        const updated = await api.updateMe({ nearbyVisible: true })
        applyProfile(updated)
        try {
          await api.updateLocation(
            position.coords.latitude,
            position.coords.longitude,
            position.coords.accuracy,
          )
          await queryClient.invalidateQueries({ queryKey: ['me'] })
        } catch {
          // Visibility is on; the first coordinate ping can be retried later.
        }
        toast.success('You are now visible on the nearby radar.')
      } catch (err) {
        setNearbyLocal(false)
        if (err instanceof Error && err.message === 'GEO_DENIED') {
          toast.error(
            'Location permission was denied. Allow location access in your browser settings to appear on the radar.',
            { duration: 7000 },
          )
        } else if (err instanceof ApiClientError) {
          toast.error(err.message)
        } else {
          toast.error('Could not get your location. Make sure location services are on.')
        }
      } finally {
        setLocationBusy(false)
      }
    } else {
      setNearbyLocal(false)
      try {
        await api.disableLocation()
        const updated = await api.updateMe({ nearbyVisible: false })
        applyProfile(updated)
        await queryClient.invalidateQueries({ queryKey: ['me'] })
        toast.success('Nearby visibility turned off. Your stored coordinates were deleted.')
      } catch (err) {
        setNearbyLocal(true)
        toast.error(err instanceof ApiClientError ? err.message : 'Could not update nearby visibility.')
      } finally {
        setLocationBusy(false)
      }
    }
  }

  async function handlePeopleVisibilityChange(next: boolean) {
    setRandomBusy(true)

    } finally {

    }
  }

/**

  onOpenChange: (open: boolean) => void

  const [audience, setAudience] = useState<StoryAudience>('public')
  const [file, setFile] = useState<File | null>(null)
  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
  const [uploadedFileId, setUploadedFileId] = useState<string | null>(null)
  const [uploading, setUploading] = useState(false)
  const [creating, setCreating] = useState(false)

      setFile(null)

'use client'

import { useEffect } from 'react'
import {
  collection,
  limit,
  onSnapshot,
  orderBy,
  query,
  type DocumentData,
} from 'firebase/firestore'
import { signInWithCustomToken } from 'firebase/auth'
import { toast } from 'sonner'
import { useQueryClient } from '@tanstack/react-query'
import { getFirebaseClientAuth, getFirebaseClientFirestore } from '@/lib/firebase-client'
import { useAppStore } from '@/store/app-store'
import type { ConversationSummary, MessageDTO } from '@/lib/types'

type RealtimeEvent = {
  event: string
  payload: DocumentData
}

function processEvent(
  queryClient: ReturnType<typeof useQueryClient>,
  event: string,
  payload: DocumentData,
) {
  switch (event) {
    case 'message:new': {
      const message = payload as MessageDTO
      queryClient.invalidateQueries({ queryKey: ['messages', message.conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      if (message.senderPublicId === useAppStore.getState().profile?.publicId) return
      if (
        useAppStore.getState().activeConversationId === message.conversationId &&
        document.visibilityState === 'visible'
      ) {
        void fetch(`/api/conversations/${message.conversationId}/read`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: '{}',
        }).catch((error: unknown) => console.error('[realtime] read receipt request failed:', error))
        return
      }

      const conversations = queryClient.getQueryData<{ conversations: ConversationSummary[] }>([
        'conversations',
      ])
      const conversation = conversations?.conversations.find(
        (item) => item.id === message.conversationId,
      )
      const sender = conversation
        ? conversation.type === 'group'
          ? conversation.name ?? 'Group chat'
          : conversation.other?.nickname ?? conversation.other?.displayName ?? 'Someone'
        : 'Someone'
      const preview =
        message.type === 'text'
          ? message.text ?? ''
          : message.type === 'image'
            ? 'Sent a photo'
            : message.type === 'video'
              ? 'Sent a video'
              : 'New message'
      toast(`${sender}: ${preview}`, {
        description: 'Open Chats to reply',
        action: {
          label: 'Open',
          onClick: () => useAppStore.getState().openConversation(message.conversationId),
        },
      })
      return
    }
    case 'conversation:new':
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      if (payload.conversationId !== useAppStore.getState().activeConversationId) {
        toast(payload.groupName ? `Added to â€œ${payload.groupName}â€` : 'New conversation started', {
          description: 'Open Chats to say hello.',
        })
      }
      return
    case 'conversation:updated':
    case 'conversation:deleted':
    case 'conversation:read':
    case 'message:changed':
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      if (payload.conversationId) {
        queryClient.invalidateQueries({ queryKey: ['conversation', payload.conversationId] })
        queryClient.invalidateQueries({ queryKey: ['messages', payload.conversationId] })
      }
      return
    case 'presence:update':
      queryClient.invalidateQueries({ queryKey: ['people'] })
      queryClient.invalidateQueries({ queryKey: ['nearby'] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      queryClient.invalidateQueries({ queryKey: ['contacts'] })
      return
    case 'feed:update':
      queryClient.invalidateQueries({ queryKey: ['posts'] })
      if (payload.postId) {
        queryClient.invalidateQueries({ queryKey: ['post-thread', payload.postId] })
      }
      return
    case 'buzz:notification':
      queryClient.invalidateQueries({ queryKey: ['posts'] })
      if (payload.postId) {
        queryClient.invalidateQueries({ queryKey: ['post-thread', payload.postId] })
      }
      toast(payload.title ?? 'Buzz', {
        description: payload.body ?? 'There is new activity on a post you contributed to.',
        action: {
          label: 'Open Buzz',
          onClick: () => useAppStore.getState().setTab('feed'),
        },
      })
  }
}

function attachEventListener(
  collectionPath: string[],
  onEvent: (event: RealtimeEvent) => void,
  onError: (error: Error) => void,
) {
  const collectionRef = collection(getFirebaseClientFirestore(), collectionPath.join('/'))
  const eventsQuery = query(collectionRef, orderBy('createdAt', 'desc'), limit(50))
  let initialSnapshot = true

  return onSnapshot(
    eventsQuery,
    (snapshot) => {
      if (initialSnapshot) {
        initialSnapshot = false
        return
      }
      for (const change of snapshot.docChanges()) {
        if (change.type !== 'added') continue
        const data = change.doc.data()
        if (typeof data.event !== 'string' || typeof data.payload !== 'object' || !data.payload) {
          console.error('[realtime] Ignoring malformed Firebase event:', change.doc.id)
          continue
        }
        onEvent({ event: data.event, payload: data.payload as DocumentData })
      }
    },
    onError,
  )
}

export function useFirebaseRealtime(firebaseToken: string | null | undefined) {
  const queryClient = useQueryClient()

  useEffect(() => {
    if (!firebaseToken) return

    let active = true
    let stopUserEvents: (() => void) | undefined
    let stopBroadcastEvents: (() => void) | undefined

    void (async () => {
      try {
        const auth = getFirebaseClientAuth()
        const credential = await signInWithCustomToken(auth, firebaseToken)
        if (!active) return

        const handleEvent = ({ event, payload }: RealtimeEvent) => {
          if (useAppStore.getState().socketConnected) return
          processEvent(queryClient, event, payload)
        }
        const onError = (error: Error) => {
          console.error('[realtime] Firebase event listener failed:', error)
        }

        stopUserEvents = attachEventListener(
          ['realtimeUsers', credential.user.uid, 'events'],
          handleEvent,
          onError,
        )
        stopBroadcastEvents = attachEventListener(
          ['realtimeBroadcasts'],
          handleEvent,
          onError,
        )
      } catch (error) {
        console.error('[realtime] Firebase authentication failed:', error)
      }
    })()

    return () => {
      active = false
      stopUserEvents?.()
      stopBroadcastEvents?.()
    }
  }, [firebaseToken, queryClient])
}

"use client";

import { useEffect, useRef } from 'react'
import { useQueryClient } from '@tanstack/react-query'
import { toast } from 'sonner'
import { getSocket } from '@/lib/socket-client'
import { useAppStore } from '@/store/app-store'
import type { ConversationSummary, MessageDTO } from '@/lib/types'

/**
 * Wires the realtime socket to the query cache and UI:
 * presence, incoming messages, Buzz activity, read receipts, and typing.
 */
export function useRealtime(socketToken: string | null | undefined, myPublicId: string | null) {
  const queryClient = useQueryClient()
  const activeConversationId = useAppStore((s) => s.activeConversationId)
  const closeConversation = useAppStore((s) => s.closeConversation)
  const setSocketConnected = useAppStore((s) => s.setSocketConnected)
  const setTab = useAppStore((s) => s.setTab)
  const setTyping = useAppStore((s) => s.setTyping)

  const activeRef = useRef(activeConversationId)
  const publicIdRef = useRef(myPublicId)
  useEffect(() => {
    activeRef.current = activeConversationId
  }, [activeConversationId])
  useEffect(() => {
    publicIdRef.current = myPublicId
  }, [myPublicId])

  useEffect(() => {
    if (!socketToken || !myPublicId) return
    const socket = getSocket(socketToken)

    const markRead = (conversationId: string) => {
      fetch(`/api/conversations/${conversationId}/read`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: '{}',
      }).catch(() => undefined)
    }

    const onConnect = () => {
      setSocketConnected(true)
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      queryClient.invalidateQueries({ queryKey: ['nearby'] })
    }

    const subscribeConv = (conversationId?: string) => {
      if (conversationId) socket.emit('conversation:subscribe', { conversationId })
    }

    const onDisconnect = () => setSocketConnected(false)

    const onMessageNew = (msg: MessageDTO) => {
      queryClient.invalidateQueries({ queryKey: ['messages', msg.conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })

      if (msg.senderPublicId === publicIdRef.current) return

      if (activeRef.current === msg.conversationId && document.visibilityState === 'visible') {
        markRead(msg.conversationId)
      } else {
        // Friendly toast with a content preview
        const convs = queryClient.getQueryData<{ conversations: ConversationSummary[] }>(['conversations'])
        const conv = convs?.conversations.find((c) => c.id === msg.conversationId)
        const who = conv
          ? conv.type === 'group'
            ? (conv.name ?? 'Group chat')
            : (conv.other?.nickname ?? conv.other?.displayName ?? 'Someone')
          : 'Someone'
        const preview =
          msg.type === 'text'
            ? msg.text ?? ''
            : msg.type === 'image'
              ? 'Sent a photo'
              : msg.type === 'video'
                ? 'Sent a video'
                : 'New message'
        toast(`${who}: ${preview}`, {
          description: 'Open Chats to reply',
          action: {
            label: 'Open',
            onClick: () => {
              useAppStore.getState().openConversation(msg.conversationId)
            },
          },
        })
      }
    }

    const onConversationNew = (data: { conversationId?: string; groupName?: string }) => {
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      subscribeConv(data?.conversationId)
      if (data?.conversationId !== activeRef.current) {
        toast(data?.groupName ? `Added to â€œ${data.groupName}â€` : 'New conversation started', {
          description: 'Open Chats to say hello.',
        })
      }
    }

    const onPresenceUpdate = () => {
      queryClient.invalidateQueries({ queryKey: ['people'] })
      queryClient.invalidateQueries({ queryKey: ['nearby'] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      queryClient.invalidateQueries({ queryKey: ['contacts'] })
    }

    const onConversationUpdated = (data: { conversationId?: string }) => {
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      if (data?.conversationId) {
        queryClient.invalidateQueries({ queryKey: ['conversation', data.conversationId] })
        queryClient.invalidateQueries({ queryKey: ['messages', data.conversationId] })
      }
    }

    const onConversationDeleted = (data: { conversationIds?: string[]; reason?: string }) => {
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
      if (data?.conversationIds?.includes(activeRef.current ?? '')) {
        closeConversation()
        toast(
          data.reason === 'account_deleted'
            ? 'This conversation ended â€” the other person deleted their account.'
            : 'This conversation was closed and removed.'
        )
      }
    }

    const onRead = (data: { conversationId?: string }) => {
      if (data?.conversationId) {
        queryClient.invalidateQueries({ queryKey: ['messages', data.conversationId] })
      }
    }

    const onMessageChanged = (data: { conversationId?: string }) => {
      if (!data?.conversationId) return
      queryClient.invalidateQueries({ queryKey: ['messages', data.conversationId] })
      queryClient.invalidateQueries({ queryKey: ['conversations'] })
    }

    const onTyping = (data: { conversationId?: string; typing?: boolean }) => {
      if (!data?.conversationId) return
      // Typing events don't carry identity of sender publicly; treat as "them"
      setTyping(data.conversationId, data.typing ? 'them' : null)
    }

    // Buzz feed broadcasts: new/deleted posts and new/deleted comments.
    const onFeedUpdate = (data: { reason?: string; postId?: string }) => {
      queryClient.invalidateQueries({ queryKey: ['posts'] })
      if (data?.postId) {
        queryClient.invalidateQueries({ queryKey: ['post-thread', data.postId] })
      }
    }

    const onBuzzNotification = (data: { title?: string; body?: string; postId?: string }) => {
      if (data?.postId) {
        queryClient.invalidateQueries({ queryKey: ['posts'] })

      }

/**
 * Thin typed client for the Ghostline API. All requests are same-origin and
 * cookie-authenticated (httpOnly session). Errors are normalized to
 * { code, message } so UI can show actionable messages.
 */

export class ApiClientError extends Error {
  code: string
  status: number
  retryAfterSec?: number
  constructor(status: number, code: string, message: string, retryAfterSec?: number) {
    super(message)
    this.status = status
    this.code = code
    this.retryAfterSec = retryAfterSec
  }
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  let res: Response
  try {
    res = await fetch(path, {
      ...init,
      headers: {
        ...(init?.body && !(init.body instanceof FormData) ? { 'Content-Type': 'application/json' } : {}),
        ...init?.headers,
      },
      credentials: 'same-origin',
    })
  } catch {
    throw new ApiClientError(0, 'NETWORK', 'You appear to be offline. Check your connection.')
  }
  if (!res.ok) {
    let code = 'ERROR'
    let message = `Request failed (${res.status})`
    let retryAfterSec: number | undefined
    try {
      const data = await res.json()
      if (data?.error) {
        code = data.error.code ?? code
        message = data.error.message ?? message
        retryAfterSec = data.error.retryAfterSec
      }
    } catch {
      /* non-JSON error */
    }
    throw new ApiClientError(res.status, code, message, retryAfterSec)
  }
  return res.json() as Promise<T>
}

import {

  type DocumentData,
} from 'firebase-admin/firestore'

import { ApiError, forbidden, notFound, gone } from '@/lib/api-helpers'

export type OtherParticipant = {
  id: string
  lastReadAt: Date | null
  profile: {
    publicId: string
    displayName: string
    avatarFileId: string | null
    showInPeople: boolean
  }
}

export type ConversationAccess = {
  conversation: {
    id: string
    type: string
    name: string | null
    status: string
    expirationPolicy: string
    expiresAt: Date | null
    createdById: string
    createdAt: Date
    updatedAt: Date
  }
  meParticipant: { id: string; lastReadAt: Date | null }
  others: OtherParticipant[]
  isGroup: boolean
  iBlockedThem: boolean
  theyBlockedMe: boolean
}

function asDate(value: unknown): Date | null {
  if (value instanceof Date) return value
  if (value instanceof Timestamp) return value.toDate()

  if (
    value &&
    typeof value === 'object' &&
    'toDate' in value &&
    typeof value.toDate === 'function'
  ) {
    return value.toDate()
  }

  if (typeof value === 'string' || typeof value === 'number') {
    const date = new Date(value)
    if (!Number.isNaN(date.getTime())) return date
  }

  return null
}

function requiredDate(value: unknown, field: string): Date {
  const date = asDate(value)
  if (!date) {
    throw new Error(`Invalid or missing conversation field: ${field}`)
  }
  return date
}

function conversationFromData(
  id: string,
  data: DocumentData,
): ConversationAccess['conversation'] {
  return {
    id,
    type: String(data.type ?? 'direct'),
    name: data.name ?? null,
    status: String(data.status ?? 'active'),
    expirationPolicy: String(data.expirationPolicy ?? '24h'),
    expiresAt: asDate(data.expiresAt),
    createdById: String(data.createdById ?? ''),
    createdAt: requiredDate(data.createdAt, 'createdAt'),
    updatedAt: requiredDate(data.updatedAt, 'updatedAt'),
  }
}

/** Loads a conversation and verifies that the caller is a participant. */
export async function getConversationAccess(
  conversationId: string,
  meId: string,
): Promise<ConversationAccess> {
  const firestore = getFirebaseAdminFirestore()
  const conversationRef = firestore.collection('conversations').doc(conversationId)
  const conversationSnap = await conversationRef.get()

  if (!conversationSnap.exists) {
    throw notFound('Conversation not found.')
  }

  const conversation = conversationFromData(
    conversationSnap.id,
    conversationSnap.data()!,
  )

  const participantsSnap = await conversationRef
    .collection('participants')
    .get()

  const participants = participantsSnap.docs.map((doc) => ({
    id: doc.id,
    userId: String(doc.data().userId ?? doc.id),
    lastReadAt: asDate(doc.data().lastReadAt),
  }))

  const mePart = participants.find((participant) => participant.userId === meId)

  if (!mePart) {
    throw forbidden(
      'You are not a participant of this conversation.',
      'NOT_PARTICIPANT',
    )
  }

  const isGroup = conversation.type === 'group'
  const otherParticipantRows = participants.filter(
    (participant) => participant.userId !== meId,
  )

  if (!isGroup && otherParticipantRows.length === 0) {
    throw notFound('The other participant is no longer available.')
  }

  const others: OtherParticipant[] = []

  for (const participant of otherParticipantRows) {
    const profileSnap = await firestore
      .collection('profiles')
      .doc(participant.userId)
      .get()

    if (!profileSnap.exists) continue

    const profile = profileSnap.data()!

    others.push({
      id: participant.userId,
      lastReadAt: participant.lastReadAt,
      profile: {
        publicId: String(profile.publicId ?? ''),
        displayName: String(profile.displayName ?? 'Anonymous'),
        avatarFileId: profile.avatarFileId ?? null,
        showInPeople: profile.showInPeople ?? true,
      },
    })
  }

  if (!isGroup && others.length === 0) {
    throw notFound('The other participant is no longer available.')
  }

  let iBlockedThem = false
  let theyBlockedMe = false

  if (!isGroup && others[0]) {
    const otherId = others[0].id
    const blocks = firestore.collection('blockedUsers')

    const [outgoingBlock, incomingBlock] = await Promise.all([
      blocks
        .where('blockerId', '==', meId)
        .where('blockedId', '==', otherId)
        .limit(1)
        .get(),
      blocks
        .where('blockerId', '==', otherId)
        .where('blockedId', '==', meId)
        .limit(1)
        .get(),
    ])

    iBlockedThem = !outgoingBlock.empty
    theyBlockedMe = !incomingBlock.empty
  }

  return {
    conversation,
    meParticipant: {
      id: mePart.id,
      lastReadAt: mePart.lastReadAt,
    },
    others,
    isGroup,
    iBlockedThem,
    theyBlockedMe,
  }
}

/** Rejects sending messages to ended, expired, or blocked direct conversations. */
export function assertCanSend(access: ConversationAccess) {
  if (access.conversation.status !== 'active') {
    throw gone(
      'This conversation has ended and no longer accepts messages.',
      'CONVERSATION_ENDED',
    )
  }

  if (isConversationExpired(access.conversation)) {
    throw gone(
      'This conversation has expired and no longer accepts messages.',
      'CONVERSATION_EXPIRED',
    )
  }

  if (
    !access.isGroup &&
    (access.iBlockedThem || access.theyBlockedMe)
  ) {
    throw forbidden(
      'Messaging is unavailable between blocked users.',
      'BLOCKED',
    )
  }
}

export async function publishFirebaseUserEvent(
  userIds: string[],
  event: string,
  payload: unknown,
): Promise<void> {
  const firestore = getFirebaseAdminFirestore()
  const uniqueIds = [...new Set(userIds)]

  for (let offset = 0; offset < uniqueIds.length; offset += 400) {
    const batch = firestore.batch()

    for (const userId of uniqueIds.slice(offset, offset + 400)) {
      const eventRef = firestore
        .collection('realtimeUsers')
        .doc(userId)
        .collection('events')
        .doc()

      batch.create(eventRef, {
        event,
        payload,
        createdAt: FieldValue.serverTimestamp(),
        expireAt: new Date(Date.now() + 24 * 60 * 60 * 1000),
      })
    }

    await batch.commit()
  }
}

/**

 * mapping, and hot ranking. This module uses Firestore instead of Prisma.

import { getFirebaseAdminFirestore } from '@/lib/firebase-admin'

import type { PostDTO, PostCommentDTO, PostAuthorDTO } from '@/lib/types'

export const POST_MAX_LEN = 300
export const COMMENT_MAX_LEN = 300

/** Matches the People directory's freshness window. */
const ONLINE_GRACE_MS = 75_000

/** How many recent posts are considered for the hot ranking. */
export const HOT_CANDIDATES = 150
export const FEED_TAKE = 100

type AuthorRow = {
  publicId: string
  displayName: string
  avatarFileId: string | null
  presence: { isOnline: boolean; lastHeartbeatAt: Date } | null
}

export function toPostAuthor(a: AuthorRow, isAnonymous = false): PostAuthorDTO {
  if (isAnonymous) {
    return {
      publicId: 'anonymous',
      displayName: 'Anonymous',
      avatarUrl: null,
      online: false,
      isAnonymous: true,
    }
  }

  const lastHeartbeat = a.presence?.lastHeartbeatAt
  const heartbeatMs = lastHeartbeat instanceof Date ? lastHeartbeat.getTime() : Number.NaN
  const online = Boolean(
    a.presence?.isOnline && heartbeatMs > Date.now() - ONLINE_GRACE_MS,
  )

  return {
    publicId: a.publicId,
    displayName: a.displayName,
    avatarUrl: avatarUrlFor(a.avatarFileId),
    online,
    isAnonymous: false,
  }
}

type PostRow = {
  id: string
  text: string
  voteScore: number
  commentCount: number
  createdAt: Date
  authorId: string
  isAnonymous: boolean
  author: AuthorRow
  votes?: { value: number }[]
}

export function toPostDTO(post: PostRow, myId: string): PostDTO {
  const my = post.votes?.[0]?.value ?? 0
  return {
    id: post.id,
    text: post.text,
    author: toPostAuthor(post.author, post.isAnonymous),
    voteScore: Number(post.voteScore ?? 0),
    commentCount: Number(post.commentCount ?? 0),
    myVote: my === 1 ? 1 : my === -1 ? -1 : 0,
    isOwn: post.authorId === myId,
    createdAt: post.createdAt.toISOString(),
  }
}

  text: string
  createdAt: Date
  authorId: string
  isAnonymous: boolean
  author: AuthorRow
}

export function toCommentDTO(c: CommentRow, myId: string): PostCommentDTO {
  return {
    id: c.id,
    text: c.text,
    author: toPostAuthor(c.author, c.isAnonymous),
    isOwn: c.authorId === myId,
    createdAt: c.createdAt.toISOString(),
  }
}

/** Reddit-style decayed score â€” higher is hotter. */
export function hotRank(p: { voteScore: number; commentCount: number; createdAt: string }): number {
  const hours = Math.max(0.5, (Date.now() - new Date(p.createdAt).getTime()) / 3_600_000)
  return (p.voteScore + p.commentCount) / Math.pow(hours + 2, 1.4)
}

/**
 * Recomputes the cached vote score from posts/{postId}/votes and writes it to
 * the parent post document. The vote endpoint maintains the score in its
 * transaction; this helper is retained for maintenance or future callers.
 */
export async function recomputeVoteScore(postId: string): Promise<number> {
  const firestore = getFirebaseAdminFirestore()
  const postRef = firestore.collection('posts').doc(postId)
  const [postSnapshot, votesSnapshot] = await Promise.all([
    postRef.get(),
    postRef.collection('votes').get(),
  ])

  const score = votesSnapshot.docs.reduce(
    (total, vote) => total + Number(vote.data().value ?? 0),
    0,
  )

  if (postSnapshot.exists) {
    await postRef.update({ voteScore: score })
  }

  return score
}

async function loadSessionByToken(token: string) {
  const firestore = getFirebaseAdminFirestore()
  const session = await getSessionByToken(token)

  if (!session) return null

  if (session.expiresAt.getTime() < Date.now()) {
    await firestore.collection('sessions').doc(token).delete()
      .catch(() => undefined)
    return null
  }

  const profileSnap = await firestore
    .collection('profiles')
    .doc(session.userId)
    .get()

  if (!profileSnap.exists) {
    await firestore.collection('sessions').doc(token).delete()
      .catch(() => undefined)
    return null
  }

  return {
    session,
    profile: profileFromData(session.userId, profileSnap.data()!),
  }
}

export async function getAuth(): Promise<AuthContext | null> {
  const jar = await cookies()
  const token = jar.get(SESSION_COOKIE)?.value

  if (!token) return null

  const loaded = await loadSessionByToken(token)
  if (!loaded) return null

  const { session, profile } = loaded
  const now = Date.now()
  const remaining = session.expiresAt.getTime() - now
  const firestore = getFirebaseAdminFirestore()

  if (remaining < SESSION_EXTEND_THRESHOLD_MS) {
    await firestore.collection('sessions').doc(token).update({
      expiresAt: Timestamp.fromMillis(now + SESSION_TTL_MS),
      lastSeenAt: FieldValue.serverTimestamp(),
    }).catch(() => undefined)
  } else if (now - session.lastSeenAt.getTime() > 10 * 60_000) {
    await firestore.collection('sessions').doc(token).update({
      lastSeenAt: FieldValue.serverTimestamp(),
    }).catch(() => undefined)
  }

  return {
    profile,
    sessionId: session.id,
    token,
  }
}

      code?: string

    err.code = 'UNAUTHENTICATED'

  }

  return auth
}

export async function createAnonymousIdentity(displayName: string) {
  const firestore = getFirebaseAdminFirestore()

  for (let attempt = 0; attempt < 5; attempt++) {
    const id = randomUUID()
    const publicId = generatePublicId()
    const inviteCode = generateInviteCode()
    const recoveryCode = generateRecoveryCode()
    const profileRef = firestore.collection('profiles').doc(id)
    const presenceRef = firestore.collection('presences').doc(id)

    try {
      const profile = await firestore.runTransaction(async (tx) => {
        const checks = await Promise.all([
          tx.get(
            firestore.collection('profiles')
              .where('publicId', '==', publicId).limit(1),
          ),
          tx.get(
            firestore.collection('profiles')
              .where('inviteCode', '==', inviteCode).limit(1),
          ),
          tx.get(
            firestore.collection('profiles')
              .where('recoveryCode', '==', recoveryCode).limit(1),
          ),
        ])

        if (checks.some((result) => !result.empty)) {
          throw new Error('IDENTITY_CODE_COLLISION')
        }

        const now = Timestamp.now()

        const data = {
          publicId,
          displayName,
          inviteCode,
          recoveryCode,
          avatarFileId: null,
          nearbyVisible: false,
          showInPeople: true,
          notifyWhenPeopleOnline: true,
          onlineNotificationAudience: 'anyone',
          defaultStoryAudience: 'public',
          createdAt: now,
          updatedAt: now,
        }

        tx.create(profileRef, data)
        tx.create(presenceRef, {
          userId: id,
          isOnline: false,
          lastHeartbeatAt: null,
          lastSeenAt: now,
        })

        return profileFromData(id, data)
      })

      return profile
    } catch (error) {
      if (
        error instanceof Error &&
        error.message === 'IDENTITY_CODE_COLLISION' &&
        attempt < 4
      ) {
        continue
      }

      console.error('[session] Failed to create Firebase identity:', error)
      throw error
    }
  }

  throw new Error('Could not generate a unique identity after five attempts.')
}

/**

 *   BUILD time
 *   â€” NEXT_PUBLIC_ vars are inlined into the client bundle.

 *
 * The server serves engine.io under /rt/socket.io (see mini-services/realtime
 * service.ts) so one public port can also expose /healthz and /notify.
 */
export function getSocket(token: string): Socket {
  if (socket && currentToken === token) return socket
  if (socket) {
    socket.disconnect()
    socket = null
  }
  currentToken = token
  const realtimeUrl = process.env.NEXT_PUBLIC_REALTIME_URL
  socket = io(realtimeUrl || '/?XTransformPort=3003', {
    path: '/rt/socket.io',
    auth: { token },
    transports: ['websocket', 'polling'],
    forceNew: true,
    reconnection: true,
    reconnectionAttempts: Infinity,
    reconnectionDelay: 1000,
    reconnectionDelayMax: 5000,
    timeout: 10000,
  })
  return socket
}

  socket?.disconnect()
  socket = null
  currentToken = null
}

"use client";

import { create } from 'zustand'
import type { OwnProfile } from '@/lib/types'

export type Tab = 'nearby' | 'people' | 'feed' | 'stories' | 'chats' | 'profile'

type TypingState = Record<string, { publicId: string; at: number }>

type AppState = {
  /** null = still bootstrapping; undefined = checked and no identity */
  profile: OwnProfile | null
  profileChecked: boolean
  setProfile: (p: OwnProfile | null) => void

  socketConnected: boolean
  setSocketConnected: (v: boolean) => void

  tab: Tab
  setTab: (t: Tab) => void

  /** When set, the chat view is open (stacked above the current tab) */
  activeConversationId: string | null
  openConversation: (id: string) => void
  closeConversation: () => void

  /** id of the pending invite code from ?invite= query param */
  pendingInviteCode: string | null
  setPendingInviteCode: (code: string | null) => void

  /** Add-to-Home-Screen popup: driven by auto-triggers + the Profile "Install app" row */
  installPromptOpen: boolean
  setInstallPromptOpen: (open: boolean) => void

  /** live typing indicators: conversationId â†’ who/when */
  typing: TypingState
  setTyping: (conversationId: string, publicId: string | null) => void
}

export const useAppStore = create<AppState>((set) => ({
  profile: null,
  profileChecked: false,
  setProfile: (profile) => set({ profile, profileChecked: true }),

  socketConnected: false,
  setSocketConnected: (socketConnected) => set({ socketConnected }),

  tab: 'people',
  // Switching tabs from the nav always pops an open chat view (iOS-style:
  // tapping a tab while pushed into a conversation returns to that tab).
  setTab: (tab) => {
    if (typeof window !== 'undefined') {
      try {
        window.sessionStorage.removeItem('gl-active-conv')
      } catch {
        /* private mode */
      }
    }
    set({ tab, activeConversationId: null })
  },

  activeConversationId: null,
  openConversation: (activeConversationId) => {
    // Remember the open chat so a refresh restores the conversation view
    // (history itself is server-side and survives refreshes regardless).
    if (typeof window !== 'undefined') {
      try {
        window.sessionStorage.setItem('gl-active-conv', activeConversationId)
      } catch {
        /* private mode */
      }
    }
    set({ activeConversationId })
  },
  closeConversation: () => {
    if (typeof window !== 'undefined') {
      try {
        window.sessionStorage.removeItem('gl-active-conv')
      } catch {
        /* private mode */
      }
    }
    set({ activeConversationId: null })
  },

  pendingInviteCode: null,
  setPendingInviteCode: (pendingInviteCode) => set({ pendingInviteCode }),

  installPromptOpen: false,
  setInstallPromptOpen: (installPromptOpen) => set({ installPromptOpen }),

  typing: {},
  setTyping: (conversationId, publicId) =>
    set((state) => {
      const typing = { ...state.typing }
      if (publicId) typing[conversationId] = { publicId, at: Date.now() }
      else delete typing[conversationId]
      return { typing }
    }),
}))
