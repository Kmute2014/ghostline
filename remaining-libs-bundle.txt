/**
 * Shared helpers for the Buzz feed (text-only posts Ã  la Jodel):
 * length limits, author/post/comment â†’ DTO mapping, and the "hot" ranking
 * used by the feed (score-weighted with time decay, computed over the most
 * recent slice of posts â€” plenty for a single-node deployment).
 */
import { db } from '@/lib/db'
import { avatarUrlFor } from '@/lib/dto'
import type { PostDTO, PostCommentDTO, PostAuthorDTO } from '@/lib/types'

export const POST_MAX_LEN = 300
export const COMMENT_MAX_LEN = 300

/** Matches the People directory's freshness window. */
const ONLINE_GRACE_MS = 75_000

/** How many recent posts are considered for the hot ranking. */
export const HOT_CANDIDATES = 150
export const FEED_TAKE = 100

type AuthorRow = {
  publicId: string
  displayName: string
  avatarFileId: string | null
  presence: { isOnline: boolean; lastHeartbeatAt: Date } | null
}

export function toPostAuthor(a: AuthorRow, isAnonymous = false): PostAuthorDTO {
  if (isAnonymous) {
    return {
      publicId: 'anonymous',
      displayName: 'Anonymous',
      avatarUrl: null,
      online: false,
      isAnonymous: true,
    }
  }
  const online = Boolean(
    a.presence?.isOnline && a.presence.lastHeartbeatAt.getTime() > Date.now() - ONLINE_GRACE_MS
  )
  return {
    publicId: a.publicId,
    displayName: a.displayName,
    avatarUrl: avatarUrlFor(a.avatarFileId),
    online,
    isAnonymous: false,
  }
}

type PostRow = {
  id: string
  text: string
  voteScore: number
  commentCount: number
  createdAt: Date
  authorId: string
  isAnonymous: boolean
  author: AuthorRow
  votes?: { value: number }[]
}

export function toPostDTO(post: PostRow, myId: string): PostDTO {
  const my = post.votes?.[0]?.value ?? 0
  return {
    id: post.id,
    text: post.text,
    author: toPostAuthor(post.author, post.isAnonymous),
    voteScore: post.voteScore,
    commentCount: post.commentCount,
    myVote: my === 1 ? 1 : my === -1 ? -1 : 0,
    isOwn: post.authorId === myId,
    createdAt: post.createdAt.toISOString(),
  }
}

type CommentRow = {
  id: string
  text: string
  createdAt: Date
  authorId: string
  isAnonymous: boolean
  author: AuthorRow
}

export function toCommentDTO(c: CommentRow, myId: string): PostCommentDTO {
  return {
    id: c.id,
    text: c.text,
    author: toPostAuthor(c.author, c.isAnonymous),
    isOwn: c.authorId === myId,
    createdAt: c.createdAt.toISOString(),
  }
}

/** Prisma `where` that excludes posts/comments authored by block-paired users. */
export function notBlockedAuthor(meId: string) {
  return {
    author: {
      blocksReceived: { none: { blockerId: meId } },
      blocksMade: { none: { blockedId: meId } },
    },
  }
}

/** Reddit-style decayed score â€” higher is hotter. */
export function hotRank(p: { voteScore: number; commentCount: number; createdAt: string }): number {
  const hours = Math.max(0.5, (Date.now() - new Date(p.createdAt).getTime()) / 3_600_000)
  return (p.voteScore + p.commentCount) / Math.pow(hours + 2, 1.4)
}

/** Recomputes the cached vote score from the PostVote rows (SQLite aggregate). */
export async function recomputeVoteScore(postId: string): Promise<number> {
  const agg = await db.postVote.aggregate({ where: { postId }, _sum: { value: true } })
  const score = agg._sum.value ?? 0
  await db.post.update({ where: { id: postId }, data: { voteScore: score } }).catch(() => undefined)
  return score
}

/**
 * Server-side Web Push (VAPID) helper for the Next.js API routes.
 *
 * NOTE: the realtime mini-service has its own copy of this logic (it knows
 * exactly which users have live sockets). This module is used by HTTP-only
 * paths, e.g. /api/push/test.
 */
import webpush from 'web-push'
import { db } from '@/lib/db'

export type PushPayload = {
  title: string
  body: string
  tag?: string
  /** Deep-link path opened when the notification is tapped, e.g. /?chat=<id> */
  url?: string
}

let configured: boolean | null = null

/** Configures web-push once; returns false when VAPID env is missing. */
export function isPushConfigured(): boolean {
  if (configured === null) {
    const publicKey = process.env.VAPID_PUBLIC_KEY
    const privateKey = process.env.VAPID_PRIVATE_KEY
    if (publicKey && privateKey) {
      try {
        webpush.setVapidDetails(
          process.env.VAPID_SUBJECT ?? 'mailto:hello@ghostline.app',
          publicKey,
          privateKey
        )
        configured = true
      } catch (err) {
        console.error('[push] invalid VAPID configuration:', err)
        configured = false
      }
    } else {
      configured = false
    }
  }
  return configured
}

export function getPublicKey(): string | null {
  return isPushConfigured() ? (process.env.VAPID_PUBLIC_KEY ?? null) : null
}

/** Hard cap per user â€” oldest subscriptions are dropped beyond this. */
const MAX_SUBSCRIPTIONS_PER_USER = 8

export async function sendPushToUser(userId: string, payload: PushPayload): Promise<number> {
  if (!isPushConfigured()) return 0
  const subs = await db.pushSubscription
    .findMany({ where: { userId }, orderBy: { createdAt: 'desc' } })
    .catch(() => [])
  if (subs.length === 0) return 0

  let delivered = 0
  const staleIds: string[] = []

  await Promise.allSettled(
    subs.map(async (sub) => {
      try {
        await webpush.sendNotification(
          {
            endpoint: sub.endpoint,
            keys: { p256dh: sub.p256dh, auth: sub.auth },
          },
          JSON.stringify(payload),
          { TTL: 3600 }
        )
        delivered += 1
      } catch (err) {
        const status = (err as { statusCode?: number }).statusCode
        if (status === 404 || status === 410) staleIds.push(sub.id)
        else console.warn(`[push] delivery failed (${status ?? 'network'}) endpoint=${sub.endpoint.slice(0, 48)}â€¦`)
      }
    })
  )

  if (staleIds.length > 0) {
    await db.pushSubscription
      .deleteMany({ where: { id: { in: staleIds }, userId } })
      .catch(() => undefined)
  }
  return delivered
}

import { PrismaClient } from '@prisma/client'
import { IpAddressTypes } from '@google-cloud/cloud-sql-connector'

const globalForPrisma = globalThis as unknown as {
  prisma: PrismaClient | undefined
  cloudSqlConnector: import('@google-cloud/cloud-sql-connector').Connector | undefined
}

async function createPrismaClient(): Promise<PrismaClient> {
  const instanceConnectionName = process.env.CLOUD_SQL_CONNECTION_NAME
  const log = ['error', 'warn'] as const

  if (!instanceConnectionName) {
    return new PrismaClient({ log: [...log] })
  }

  const databaseUrl = process.env.DATABASE_URL
  if (!databaseUrl) {
    throw new Error('DATABASE_URL must be set when CLOUD_SQL_CONNECTION_NAME is configured.')
  }

  const parsedUrl = new URL(databaseUrl)
  const connectorModule = await import('@google-cloud/cloud-sql-connector')
  const { Pool } = await import('pg')
  const { PrismaPg } = await import('@prisma/adapter-pg')
  const connector = new connectorModule.Connector()
  const connectorOptions = await connector.getOptions({
    instanceConnectionName,
    ipType: IpAddressTypes.PUBLIC,
  })

  globalForPrisma.cloudSqlConnector = connector

  const pool = new Pool({
    ...connectorOptions,
    user: decodeURIComponent(parsedUrl.username),
    password: decodeURIComponent(parsedUrl.password),
    database: decodeURIComponent(parsedUrl.pathname.slice(1)),
    max: 5,
  })

  return new PrismaClient({
    adapter: new PrismaPg(pool),
    log: [...log],
  })
}

export const db = globalForPrisma.prisma ?? (await createPrismaClient())

globalForPrisma.prisma = db
